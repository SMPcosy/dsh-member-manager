### 4.3 ESC 关闭：window 守卫 + 打开时注册/关闭时摘除
```js
// lib/client.js:1234-1244
useEffect(() => {
  if (deptOrg === null) return undefined;
  if (typeof window === "undefined" || typeof window.addEventListener !== "function") return undefined;
  function onKey(e) {
    if (e.key === "Escape") setDeptOrg(null);
  }
  window.addEventListener("keydown", onKey);
  return () => window.removeEventListener("keydown", onKey);
}, [deptOrg]);
```

### 4.4 弹层结构：遮罩 target 自关 + 卡片 stopPropagation + 关闭按钮
```js
// lib/client.js:1475-1504（关键节选）
deptOrg !== null
  ? h("div", {
        style: S.modal,
        onClick: (e) => { if (e.target === e.currentTarget) setDeptOrg(null); },
      },
      h("div", {
        style: Object.assign({}, S.modalPanel, { width: "min(1080px, 94vw)", maxHeight: "86vh", overflow: "auto" }),
        onClick: (e) => e.stopPropagation(),
      },
        h("div", { /* 标题栏 */ },
          h("span", { fontWeight: 600, fontSize: 14 }, t("tabOrg")),
          h("span", { fontSize: 11.5, opacity: 0.6 }, t("dpModalHint")),
          h("span", { marginLeft: "auto" },
            h("button", { style: S.btnSm, onClick: () => setDeptOrg(null) }, t("cpClose")),
          ),
        ),
        deptOrgPage,
      ),
    )
  : null,
```
- 遮罩 onClick 用 `e.target === e.currentTarget` 判断「点到了遮罩本身」，点击卡片或卡片内按钮不触发（冒泡被卡片 stopPropagation 拦截）；
- 卡片 onClick 统一 stopPropagation，比只在按钮上 stopPropagation 更稳健（卡片内任意子元素点击都不会冒到遮罩）；
- 关闭按钮 onClick 直接 `setDeptOrg(null)`，不依赖事件冒泡；
- 卡片宽度 `min(1080px, 94vw)` + `maxHeight: 86vh; overflow: auto`，避免弹层超屏。
